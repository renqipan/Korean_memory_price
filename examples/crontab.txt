# Example cron schedule. Replace the Python path after checking `command -v python3`.
# Times use the machine's local timezone. Load DATA_GO_KR_SERVICE_KEY from a
# permission-restricted environment file; do not paste the key into this file.
# Create output/logs before installing the entry.

# Weekly refresh, every Monday 08:10. mkdir is used as a macOS-compatible lock.
10 8 * * 1 cd /Users/renqi/Documents/Codex/Korean_DRAM_price && mkdir /tmp/korean-memory-price.lock 2>/dev/null && trap 'rmdir /tmp/korean-memory-price.lock' EXIT && /usr/local/bin/python3 -B -m korean_memory_price run --start 202301 --end latest --outdir output/memory >> output/logs/monthly.log 2>&1
